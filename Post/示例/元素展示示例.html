元素展示示例

#include <iostream>
#include <list>
int main() {
   std::list<int> numbers = {10, 20, 30};
   numbers.push_back(40);
   numbers.push_front(5);
   std::cout << "List elements: ";
   for (const auto& num : numbers) {
       std::cout << num << " ";
   }
   std::cout << std::endl;
   numbers.pop_back();
   numbers.pop_front();
   std::cout << "After modifications: ";
   for (const auto& num : numbers) {
       std::cout << num << " ";
   }
   return 0;
   //test code...
}
